rules_version = '2';

// Mathkitty security rules.
// Paste this whole file into Firebase Console > Firestore Database >
// Rules, then press Publish.
//
// The short version: you can only ever write your own data, everyone
// signed in can read the leaderboard, and nobody can read anyone
// else's private progress.

service cloud.firestore {
  match /databases/{database}/documents {

    // Private progress: attempts log, flags, syllabus marks.
    // Only the owner can read or write it.
    match /users/{uid} {
      allow read, write: if request.auth != null && request.auth.uid == uid;
    }

    
    match /leaderboard/{uid} {
      allow read: if request.auth != null;
      allow write: if request.auth != null
                   && request.auth.uid == uid
                   && request.resource.data.username is string
                   && request.resource.data.username.matches('^[a-z0-9_]{3,16}$')
                   && request.resource.data.solved is int
                   && request.resource.data.solved >= 0
                   && request.resource.data.solved <= 2000
                   && request.resource.data.xp is int
                   && request.resource.data.xp == request.resource.data.solved * 10
                   && request.resource.data.streakDays is int
                   && request.resource.data.streakDays >= 0
                   && request.resource.data.streakDays <= 1000;
    }

    // Class question counters. Signed-in users may only nudge a counter
    // up by one at a time, so nobody can rewrite the class statistics.
    // The documents hold totals only, never who answered what.
    match /qstats/{qid} {
      allow read: if request.auth != null;
      allow create: if request.auth != null
                    && request.resource.data.attempts == 1
                    && request.resource.data.get('correct', 0) <= 1;
      allow update: if request.auth != null
                    && request.resource.data.attempts == resource.data.attempts + 1
                    && request.resource.data.get('correct', 0) >= resource.data.get('correct', 0)
                    && request.resource.data.get('correct', 0) <= resource.data.get('correct', 0) + 1;
    }

    
    match /materials/{unitId} {
      allow read: if request.auth != null;
      allow write: if request.auth != null
                   && request.auth.uid == 'H6WpiM7f9XR3N4Sp6PaNFfEa4Af1';
    }

    // Anything else is closed.
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
